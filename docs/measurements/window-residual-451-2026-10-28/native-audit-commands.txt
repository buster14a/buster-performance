# audit-v1
tailscale ssh david@benchpress 'EXCLUSIVE_OWNER=codex-451-residual EXCLUSIVE_TITLE='\''Issue451 final exact source/headers/status/phase audit'\'' EXCLUSIVE_LINK=https://github.com/buster14a/buster/issues/451 /home/david/buster-fast-bench/exclusive.sh taskset -c 2 bash -c '\''set -e
RUN='\''\'\'''\''/home/david/buster-fast-bench/unclaimed-perf-20261028/451-residual-e7f89091'\''\'\'''\''
SRC='\''\'\'''\''/home/david/buster-fast-bench/unclaimed-perf-20261028/integrated-e7-672-clang2314/source'\''\'\'''\''
BIN='\''\'\'''\''/home/david/buster-fast-bench/unclaimed-perf-20261028/integrated-e7-672-clang2314/compiler-native'\''\'\'''\''
TOOL='\''\'\'''\''/home/david/buster-fast-bench/unclaimed-perf-20261028/toolchain-clang2314/buster/benchpress-clang/efdd1dbd089b99bfcfcde7b8eb1236163f2e88df5a981cca4f783ec9cb5e0378'\''\'\'''\''
export PATH="$TOOL/bin:$PATH"
export LD_LIBRARY_PATH="$TOOL/lib${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"
export SOURCE_DATE_EPOCH=1
export BUSTER_CLANG_RESOURCE_DIR="$("$TOOL/bin/clang" -print-resource-dir)"
unset MAKEFLAGS MFLAGS GNUMAKEFLAGS
cd "$SRC"
date -u +started_utc=%FT%TZ
for input in sqlite main long-lines spliced; do
  "$RUN/harness/phase451" "$RUN/native/phase-$input-windows1.stdout" "$RUN/native/phase-$input-whole1.stdout" "$RUN/native/phase-$input-whole2.stdout" "$RUN/native/phase-$input-windows2.stdout" > "$RUN/native/phase-$input.summary"
  cat "$RUN/native/phase-$input.summary"
done
for value in "$RUN/native/"*.status; do
  name=$(basename "$value")
  task_status=$(cat "$value")
  if test "$name" = gate-preprocessor.status; then test "$task_status" -eq 1; else test "$task_status" -eq 0; fi
  printf '\''\'\'''\''%s %s\n'\''\'\'''\'' "$name" "$task_status"
done > "$RUN/native/status-inventory.txt"
test "$(wc -l < "$RUN/native/status-inventory.txt")" -eq 45
cat "$RUN/native/status-inventory.txt"
sha256sum --check --quiet "$RUN/harness/source-check.sha256"
sha256sum --check --quiet "$RUN/main-native-dependencies.sha256"
cmp "$BIN" "$RUN/header-control/build/compiler"
cmp "$RUN/native/preflight-sqlite-E-windows.i" "$RUN/native/preflight-sqlite-E-whole.i"
cmp "$RUN/native/preflight-main-E-windows.i" "$RUN/native/preflight-main-E-whole.i"
sha256sum "$BIN" "$RUN/header-control/build/compiler" "$TOOL/bin/clang" "$RUN/main-native-predefines.h" "$RUN/main-native-dependencies.sha256" "$RUN/harness/"*.c "$RUN/inputs/"*.c "$RUN/native/"*.i > "$RUN/native/identity-after.sha256"
cat "$RUN/native/identity-after.sha256"
find "$RUN" -maxdepth 2 -type f -printf '\''\'\'''\''%P\n'\''\'\'''\'' | sort
date -u +finished_utc=%FT%TZ'\''' > work/evidence-451-residual/native-final-audit.log 2>&1
