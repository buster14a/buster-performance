set -e
date -u +%Y-%m-%dT%H:%M:%SZ >receipts/comparison-start.txt
objcopy --version >receipts/objcopy-version.txt
sha256sum main/build/compiler integrated/build/compiler >receipts/executables-before-inspection.sha256
for name in main integrated; do
  objcopy --dump-section .text=receipts/$name.text --dump-section .rodata=receipts/$name.rodata --dump-section .data.rel.ro=receipts/$name.data-rel-ro $name/build/compiler receipts/$name.inspection.elf
done
sha256sum receipts/main.text receipts/integrated.text >receipts/text.sha256
sha256sum receipts/main.rodata receipts/integrated.rodata >receipts/rodata.sha256
sha256sum receipts/main.data-rel-ro receipts/integrated.data-rel-ro >receipts/data-rel-ro.sha256
wc -c receipts/main.text receipts/integrated.text receipts/main.rodata receipts/integrated.rodata receipts/main.data-rel-ro receipts/integrated.data-rel-ro >receipts/section-sizes.txt
if cmp main/build/compiler integrated/build/compiler >receipts/full-executable-cmp.txt 2>&1; then full_cmp=0; else full_cmp=$?; fi
if cmp receipts/main.text receipts/integrated.text >receipts/text-cmp.txt 2>&1; then text_cmp=0; else text_cmp=$?; fi
if cmp receipts/main.rodata receipts/integrated.rodata >receipts/rodata-cmp.txt 2>&1; then rodata_cmp=0; else rodata_cmp=$?; fi
if cmp receipts/main.data-rel-ro receipts/integrated.data-rel-ro >receipts/data-rel-ro-cmp.txt 2>&1; then data_rel_ro_cmp=0; else data_rel_ro_cmp=$?; fi
sha256sum main/build/compiler integrated/build/compiler >receipts/executables-after-inspection.sha256
if cmp receipts/executables-before-inspection.sha256 receipts/executables-after-inspection.sha256 >receipts/originals-unchanged-cmp.txt 2>&1; then originals_cmp=0; else originals_cmp=$?; fi
printf "full_executable_cmp_exit=%s\ntext_cmp_exit=%s\nrodata_cmp_exit=%s\ndata_rel_ro_cmp_exit=%s\noriginals_unchanged_cmp_exit=%s\n" "$full_cmp" "$text_cmp" "$rodata_cmp" "$data_rel_ro_cmp" "$originals_cmp" >receipts/comparison.txt
if diff -u receipts/main-source.sha256 receipts/integrated-source.sha256 >receipts/source-manifest.diff; then source_manifest_diff=0; else source_manifest_diff=$?; fi
printf "source_manifest_diff_exit=%s\n" "$source_manifest_diff" >>receipts/comparison.txt
cmp receipts/main-clang-binary.sha256 receipts/integrated-clang-binary.sha256
cmp receipts/main-linker.txt receipts/integrated-linker.txt
date -u +%Y-%m-%dT%H:%M:%SZ >receipts/comparison-end.txt
cat receipts/comparison.txt
test "$originals_cmp" = 0
if test "$full_cmp" = 0 && test "$text_cmp" = 0 && test "$rodata_cmp" = 0 && test "$data_rel_ro_cmp" = 0; then printf "0\n" >receipts/identity-result.status; else printf "1\n" >receipts/identity-result.status; fi
