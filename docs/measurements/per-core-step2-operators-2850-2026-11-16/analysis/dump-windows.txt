The windows' dumps (windows-body.txt.gz, windows-packed.txt.gz) are sqlite3.c's windows as the first build of the
operators' kernel made them (results/o1's patch), written by this temporary code at the start of node_shadow_facts's
`else` branch (c_check_node_census.c), with BUSTER_FACTS_DUMP set to the output's path:

    if (getenv("BUSTER_FACTS_DUMP") != NULL)
    {
        FILE* dump = fopen(getenv("BUSTER_FACTS_DUMP"), "a");
        for (uint32_t window = 0; dump != NULL && window < windows; window += 1)
        {
            const uint32_t first = facts->windows[window];
            const uint32_t count = facts->windows[window + 1] - first;
            fprintf(dump, "W %u\n", count);
            for (uint32_t i = 0; i < count; i += 1)
            {
                const uint32_t w = first + i;
                const CheckOp op = (CheckOp)facts->ops[w];
                const uint32_t code = facts->kinds[w] == BUSTER_NODE_FACTS_LOOKUP ? node_facts_lookup(op) : facts->kinds[w] == BUSTER_NODE_FACTS_OPERATOR ? node_facts_operator(op) : 99;
                fprintf(dump, "%u %u %u %u %u %u %u", facts->kinds[w], code, (uint32_t)op, facts->left[w], facts->right[w], op == CHECK_OP_CONDITIONAL ? facts->inputs[w] : 0, facts->roots[w]);
                for (uint32_t k = 0; op == CHECK_OP_CALL && facts->kinds[w] == BUSTER_NODE_FACTS_LOOKUP && k < facts->counts[w]; k += 1)
                {
                    fprintf(dump, " %u", facts->arguments[facts->inputs[w] + k]);
                }
                fprintf(dump, "\n");
            }
        }
        if (dump != NULL)
        {
            fclose(dump);
        }
    }

The codes are that build's: 0 `->`, 1 `.`, 2 `*`, 3 a subscript, 4 a call, 5 a discard, 6 a row's operator, 7 `=`, 8 a
conversion, 9 a truth, 10 `&x`, 11 a cast, 12 `&&` and `||`, 13 an update, 14 the rest; 99 for a record the kernel
makes no operator of. The commands ran with the record's facts.sh options for sqlite3.c, `--facts --facts-operators
--passes 0`, and for the packed windows `--facts-packed`, by the container's build (the windows don't depend on the
host).
