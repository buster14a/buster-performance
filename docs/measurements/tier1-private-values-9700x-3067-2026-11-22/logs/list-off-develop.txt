== Measure the frames and keep the frame pointer on request
# #177: measure-frames reads each function's frame from an object's code, and -fno-omit-frame-pointer keeps one in every
# function, by Tier 0 and by Tier 1, a leaf among them, which -momit-leaf-frame-pointer leaves out; without the option
# neither tier keeps one. The depth follows the control flow (#1415's review): an early return's epilogue takes nothing
# from the deeper path after it (168 bytes, where the instructions' address order would say 128), a jump table's target
# takes its indirect jump's depth (assumed), and two paths that join at different depths are unresolved and left out of
# the comparison. rbp's depth is each path's too: two arms that set it at different depths, each with its own `leave`,
# make an 80-byte frame, resolved. A command without files prints its usage and fails.
directory=$(mktemp -d)
printf 'int g(char*);\nint f(int x) { char b[4096]; b[x & 4095] = 1; return g(b); }\nint leaf(int x) { return x + 1; }\n' > "$directory/f.c"
"$BUSTER_COMPILER" -c -o "$directory/t0.o" "$directory/f.c"

== Link Tier 1's literal pools with clang's and merge their equal entries
# #2744: a vector constant no broadcast makes is a literal its function's code loads rip-relative, which -c's object puts
# in the unit's pool: .rodata.cst16 here, mergeable (SHF_MERGE, entry size 16) and aligned to 16, its load relocated
# against a local symbol. Two units with the same constant, by strict Tier 1 for the baseline (any host runs them), linked
# with a main by clang: the program prints clang -O0's output, and the linker merged the two entries into one.
objects=$(mktemp -d)
printf '%s\n' '#include <emmintrin.h>' 'long long a(long long x)' '{' \
    '    return _mm_cvtsi128_si64(_mm_add_epi32(_mm_cvtsi64_si128(x), _mm_set_epi32(0x0badf00d, 0x0ddba11, 0x5eed1e55, 0x7e57ab1e)));' '}' > "$objects/a.c"
sed 's/long long a(/long long b(/' "$objects/a.c" > "$objects/b.c"
printf '%s\n' '#include <stdio.h>' 'long long a(long long x);' 'long long b(long long x);' \
    'int main(void) { printf("%llx %llx\n", (unsigned long long)a(1), (unsigned long long)b(2)); return 0; }' > "$objects/main.c"
"$BUSTER_COMPILER" --tier1 --tier1-strict -march=x86-64 -c -o "$objects/a.o" "$objects/a.c"
"$BUSTER_COMPILER" --tier1 --tier1-strict -march=x86-64 -c -o "$objects/b.o" "$objects/b.c"
readelf -SW "$objects/a.o" | grep -Eq '\.rodata\.cst16 +PROGBITS +0+ +[0-9a-f]+ +0+10 +10 +AM +0 +0 +16$'
objdump -dr "$objects/a.o" | grep -q 'R_X86_64_PC32[[:space:]]*\.Lbuster_literal_0-0x4$'
clang -o "$objects/tier1" "$objects/main.c" "$objects/a.o" "$objects/b.o"
