== Check unity fixture ownership without rebuilding
# The existing compiler's dependency graph keeps private fixtures, arbitrary nested includes and test headers in
# one main unit. Exact shared exceptions and shared-root structure may not grow silently; stale/incomplete records
# fail closed. Internally guarded roots also retain their complete standalone preprocessing.
# It reads the build's tree, not the executable. A hosted shard that runs its row's shared compiler (BUSTER_PRODUCT,
# docs/ci.md's "Shared row products") has no such tree: the job that built the product ran this heading on its tree
# before packing it (tools/ci/hosted-product.sh ownership), and the product records that it passed.
if [ -z "${BUSTER_PRODUCT:-}" ]; then
cc -std=c17 -O2 -Wall -Wextra -Werror -o build/build-regression tools/ci/build_regression.c
build/build-regression --ownership-only "$BUSTER_COMPILER"
else
test "$(head -n 1 "$BUSTER_PRODUCT/provenance/ownership")" = passed
printf 'SKIP: unity fixture ownership, which passed on the shared product'"'"'s build tree (%s)\n' "$BUSTER_PRODUCT"
exit 77
fi

== Compile the lexer's benchmark unit by Tier 1 on one lane and on four [4 CPUs]
# #1344: with inlining, a body's code holds its inlined callees' calls of runtime functions, whatever their parts, and the
# emission's merge gave those calls their symbols only in the parts that lowered such a call. On more than one lane
# tokenize_chunks's memmove calls, inlined into the nine tokenize_c functions, went to a case of the scalar lexer's
# switch, and the object depended on the lanes. It must be the same bytes on one lane and on four (as many of the four
# as the command's CPUs allow; compiled for znver5 on any host, never run), with every one of those calls memmove's.
objects=$(mktemp -d)
lexer=(--tier1 -fno-asynchronous-unwind-tables -march=znver5 -Isrc -fwrapv -funsigned-char -fno-strict-aliasing -c src/buster/compiler/c_lexer_benchmark.c)
"$BUSTER_COMPILER" --lanes 1 "${lexer[@]}" -o "$objects/one.o"
"$BUSTER_COMPILER" --lanes 4 "${lexer[@]}" -o "$objects/four.o"
cmp "$objects/one.o" "$objects/four.o"
test "$(objdump -dr "$objects/four.o" | grep -c 'R_X86_64_PLT32.*memmove')" -ge 18
rm -r "$objects"
