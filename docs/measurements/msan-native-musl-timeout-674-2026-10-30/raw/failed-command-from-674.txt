== Compare -c's functions and objects with clang -O0 -c's on frozen musl sources
# #129: each source the checker accepts is written as -c writes it, and must define what clang -O0 -c's object
# does, but for the three documented reasons (docs/c-lowering.md); a file the backend refuses is counted with the
# feature it names. Its acceptance against clang -fsyntax-only is compared too, as for any corpus. These eight
# directories' 348 files that the backend lowers, with every explained difference of musl's (math's), must all
# be compared: a backend that came to refuse them would otherwise pass by comparing none.
unset CPATH C_INCLUDE_PATH
export SOURCE_DATE_EPOCH=1792454400
corpus=$BUSTER_CORPUS
musl="$corpus/libc/musl"
./build/compiler test-checker-corpus --clang clang --symbols-minimum 354 --no-default-config --target=x86_64-linux-gnu -march=znver5 -std=c99 -ffreestanding -nostdinc -D_XOPEN_SOURCE=700 -Os -pipe \
  -I "$musl/arch/x86_64" -I "$musl/arch/generic" -I "$corpus/musl-generated/obj/src/internal" -I "$musl/src/include" -I "$musl/src/internal" -I "$corpus/musl-generated/obj/include" -I "$musl/include" \
  "$musl/src/math" "$musl/src/string" "$musl/src/stdlib" "$musl/src/fenv" "$musl/src/legacy" "$musl/src/process" "$musl/src/signal" "$musl/src/multibyte"
